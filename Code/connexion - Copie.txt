require_once './SOURCES/function.php';

try {
    $db = new PDO(
        'mysql:host=' . Config::getBddInfo(1) . ';dbname=' . Config::getBddInfo(3),
        Config::getBddInfo(2),
        Config::getBddInfo(70, "0100101001"),
        [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
        ]
    );
} catch (PDOException $e) {
    die("Erreur de connexion à la base de données.");
}
$controller = new connexionController($db);

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';
    $remember = isset($_POST['remember']);

    if ($username !== '' && $password !== '') {
        $connexion = $controller->byUsername($username, $password);

        if(!isset($_SESSION['try']) || !is_numeric($_SESSION['try']) || $_SESSION['try']<=3){
			if ($connexion instanceof Connexion) {
				$expire = $remember ? time() + 60 * 60 * 24 * 30 : 0;

				$cookieOptions = [
					'expires'  => $expire,
					'path'     => '/',
					'secure'   => true,      // mets false si tu n'es pas en HTTPS
					'httponly' => true,
					'samesite' => 'Strict'
				];
				
				$_SESSION['user']=["token"=>$connexion->getToken(),'cid'=>$connexion->getCid(),"tokenValidity"=>$connexion->getTokenValidity()];
				if($remember){
					setcookie('token', $connexion->getToken(), $cookieOptions);
					setcookie('cid', $connexion->getCid(), $cookieOptions);
				}

				header('Location: dashboard.php');
				exit;
			}
			else{
				if(!isset($_SESSION['try']) || !is_numeric($_SESSION['try'])){$_SESSION['try']=0;}
				$_SESSION['try']=$_SESSION['try']+1;
			}
		}
		else{
			$controller->setLock($username,$_SESSION['try']);
		}
    }
}

<script>
// Focus auto sur l’identifiant
document.getElementById('username')?.focus();

function showToast(message) {
    const container = document.getElementById('toast-container');
    const toast = document.createElement('div');
    toast.className = 'toast';

    toast.innerHTML = `
        <div class="toast-icon">⚠️</div>
        <div>${message}</div>
        <div class="toast-close" onclick="closeToast(this.parentElement)">✖</div>
    `;

    container.appendChild(toast);

    setTimeout(() => closeToast(toast), 5000);
}

function closeToast(toast) {
    toast.style.animation = 'toast-out 0.25s forwards';
    setTimeout(() => toast.remove(), 250);
}
